MySQL DDL 一键转 PostgreSQL

🐬 MySQL
-- MySQL DDL
CREATE TABLE users (
  id INT AUTO_INCREMENT,
  name VARCHAR(100) NOT NULL,
  email VARCHAR(255) UNIQUE,
  salary DECIMAL(10,2),
  created_at DATETIME
    DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (id)
);
⬇️
🐘 PostgreSQL
-- PostgreSQL DDL
CREATE TABLE users (
  id SERIAL,
  name VARCHAR(100) NOT NULL,
  email VARCHAR(255) UNIQUE,
  salary DECIMAL(10,2),
  created_at TIMESTAMP
    DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (id)
);

✨ AUTO_INCREMENT↔SERIAL | TINYINT↔SMALLINT | DATETIME↔TIMESTAMP | 50+类型映射
